# Forward Command — sync/abort/complete events (DESIGN §10.3).
# All events are is_triggered_only: without it they are random-fire
# eligible on the daily pulse (vanilla: 6586 uses).

add_namespace = tac_sync
add_namespace = tac_complete
add_namespace = tac_abort
add_namespace = tac_quiet
add_namespace = tac_watchdog
add_namespace = tac_notice

# Hourly sync acknowledgement (fired by d_tac_sync_hourly).
country_event = {
	id = tac_sync.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "{\"type\":\"tac_state\",\"hour\":0,\"phase\":\"synced\"}"
	}
}

# Battle complete (fired by d_tac_end_battle). Constant receipt — console-
# set variables are unreadable by mod scripts, so no result data rides the
# event. Visible popup stating the battle ended.
country_event = {
	id = tac_complete.99
	title = tac_complete.99.t
	desc = tac_complete.99.d
	is_triggered_only = yes

	immediate = {
		log = "{\"type\":\"tac_complete\"}"
	}

	option = {
		name = tac_complete.99.o
	}
}

# Abort — fired by the Force Exit decision, or injected on the picker
# cancel. Shared cleanup + visible popup.
country_event = {
	id = tac_abort.1
	title = tac_abort.1.t
	desc = tac_abort.1.d
	is_triggered_only = yes

	immediate = {
		d_tac_clear_tactical = yes
		log = "{\"type\":\"tac_abort\"}"
	}

	option = {
		name = tac_abort.1.o
	}
}

# Quiet-state notify: the player picked a state that currently holds NO
# running battle — the external program injects `event tac_quiet.1 <tag>`.
# The popup explains; the shared cleanup resets every map decision. Logs
# tac_abort so the external listener reads it as a uniform "tactical mode
# off" signal.
country_event = {
	id = tac_quiet.1
	title = tac_quiet.1.t
	desc = tac_quiet.1.d
	is_triggered_only = yes

	immediate = {
		d_tac_clear_tactical = yes
		log = "{\"type\":\"tac_abort\"}"
	}

	option = {
		name = tac_quiet.1.o
	}
}

# Watchdog rescue (DESIGN §10.5): fired by the daily pulse after two
# consecutive days without an exe sync ack — the external program is
# absent or died mid-battle. Shared cleanup lifts the freeze; the popup
# explains and points at the Workshop page. Logs tac_abort so a
# still-listening exe reads the uniform "tactical mode off" signal.
country_event = {
	id = tac_watchdog.1
	title = tac_watchdog.1.t
	desc = tac_watchdog.1.d
	is_triggered_only = yes

	immediate = {
		d_tac_clear_tactical = yes
		log = "{\"type\":\"tac_abort\"}"
	}

	option = {
		name = tac_watchdog.1.o
	}
}

# One-time boot notice (DESIGN §10.6): fired by the on_startup on_action
# once per new game (never on save load). Plain informational popup — no
# game.log line, no exe involvement.
country_event = {
	id = tac_notice.1
	title = tac_notice.1.t
	desc = tac_notice.1.d
	is_triggered_only = yes

	option = {
		name = tac_notice.1.o
	}
}
