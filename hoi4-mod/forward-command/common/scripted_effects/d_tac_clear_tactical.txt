# Shared tactical-mode cleanup: clear the picked-state markers, lift the
# battle-state freeze (DESIGN §8.4), clear the mode flag, purge decision
# cooldowns — every map decision is immediately re-armed. Callers:
# tac_abort.1 (Force Exit / picker cancel), tac_quiet.1 (quiet-state
# notify), d_tac_end_battle.
#
# The picked state's per-target take record is engine-locked to the daily
# pulse: left alone it decays there and the state's map icon disappears
# with it. That is the intended end state — entry decisions stay dark
# after a battle until the next scan republishes from the save of the
# moment — so nothing here re-arms that record.
d_tac_clear_tactical = {
	# Clear the picked-state marker everywhere (only ever set on a picked
	# candidate state; full sweep is cheap).
	every_state = {
		set_variable = { tac_pick = 0 }
	}
	# Lift the battle-state freeze (DESIGN §8.4); a no-op when no state is
	# frozen, so every caller gets it unconditionally.
	d_tac_unfreeze_all = yes
	clr_country_flag = tac_mode_active
	# Watchdog mailboxes (§10.5) — every cleanup path leaves them empty.
	clr_country_flag = tac_alive
	clr_country_flag = tac_watchdog_warned
	# Per-battle handshake state (§3.3): the next battle starts unseen, so
	# its own program gets its own verdict.
	clr_country_flag = tac_ver_seen
	clr_country_flag = tac_ver_warned
	# Purge the ordinary-decision cooldown records so the scan row, the entry
	# row and the Force Exit row are immediately re-clickable after a reset
	# (vanilla INS/YUG/RAJ_GOE/NSB form).
	remove_decision_on_cooldown = tactical_scan
	remove_decision_on_cooldown = tactical_command_entry
	remove_decision_on_cooldown = tactical_force_exit
}
